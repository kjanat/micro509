JwsHeader ​

A protected header. Every member the token carried stays reachable through members.

ts
interface JwsHeader<TAlgorithm extends string> {
	readonly alg: TAlgorithm;
	readonly typ?: string;
	readonly kid?: string;
	readonly cty?: string;
	readonly members: JsonObject;
}

Properties

  • readonly alg: TAlgorithm — The alg member.
  • readonly typ?: string — The typ member, when present.
  • readonly kid?: string — The kid member, when present.
  • readonly cty?: string — The cty member, when present.
  • readonly members: JsonObject — Every header member as the token carried it.
